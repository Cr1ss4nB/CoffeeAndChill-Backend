      "full_name": "John Doe",
      "email": "john@example.com",
      "password": "secure123",
      "role": "waiter",
      "phone": "555-1234"
  })

  Inventory Service Usage:
  ───────────────────────

      quantity=100.0, notes="Initial stock"
  )
  stock = get_current_stock(session, 5)

      payment_method="CARD", system_user_id=1
  )
  process_payment(session, payment.payment_id)
